<!-- SCRIPTS: TYPING EFFECT & REVEAL ON SCROLL -->
<script>
    document.addEventListener("DOMContentLoaded", function () {
        // --- 1. TYPING EFFECT SCRIPT ---
        const text = "Welcome to SUA Student Housing";
        const typedTextElement = document.getElementById("typed-text");
        
        let index = 0;
        let isDeleting = false;
        
        const typeSpeed = 100;
        const deleteSpeed = 40;
        const holdWordTime = 90000; // Dakika 1.5

        function typeEffect() {
            if (isDeleting) {
                typedTextElement.textContent = text.substring(0, index - 1);
                index--;
            } else {
                typedTextElement.textContent = text.substring(0, index + 1);
                index++;
            }

            if (!isDeleting && index === text.length) {
                setTimeout(() => { 
                    isDeleting = true; 
                    typeEffect(); 
                }, holdWordTime);
                return;
            } 
            else if (isDeleting && index === 0) {
                isDeleting = false;
            }

            const currentSpeed = isDeleting ? deleteSpeed : typeSpeed;
            setTimeout(typeEffect, currentSpeed);
        }

        typeEffect();

        // --- 2. REVEAL ON SCROLL SCRIPT (JS ANIMATION) ---
        const reveals = document.querySelectorAll(".reveal");

        const revealOnScroll = function () {
            for (let i = 0; i < reveals.length; i++) {
                const windowHeight = window.innerHeight;
                const elementTop = reveals[i].getBoundingClientRect().top;
                const elementVisible = 100; // Distance in px before revealing

                if (elementTop < windowHeight - elementVisible) {
                    reveals[i].classList.add("active");
                }
            }
        };

        window.addEventListener("scroll", revealOnScroll);
        revealOnScroll(); // Run once on load
    });
</script>
