from django import forms
from django.contrib.auth.models import User
from django.contrib.auth.forms import UserCreationForm

class UserRegisterForm(UserCreationForm):
    first_name = forms.CharField(required=True, label="First Name ")
    last_name = forms.CharField(required=True, label="Last Name ")
    # Tunalazimisha email iandikwe na mtumiaji
    email = forms.EmailField(required=True, label="Email Address")
    # phone_number = forms.CharField(required=True, label="Phone Number (+2557XXX)")
    class Meta:
        model = User
        # Hapa tunaambia Django itengeneze form yenye username na email pekee (password zitajazwa chini yake kiotomatiki)
        fields = ['username', 'first_name', 'last_name', 'email' ]
